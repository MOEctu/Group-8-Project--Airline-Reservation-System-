// Read URL parameters
const params = new URLSearchParams(window.location.search);

const booking = params.get("booking") || "N/A";
const name = params.get("name") || "N/A";
const flight = params.get("flight") || "N/A";
const seat = params.get("seat") || "N/A";

document.getElementById("flightNumber").textContent = flight;
document.getElementById("seatPreference").textContent = seat;

// Fill boarding pass

document.getElementById("bpFlight").textContent = flight;
document.getElementById("bpSeat").textContent = seat;

// Generate QR Code
const qr = new QRious({
    element: document.getElementById("qrCode"),
    size: 180,
    value: `Passenger: ${name}\nFlight: ${flight}\nSeat: ${seat}\nBooking: ${booking}`
});

// Download Boarding Pass
function downloadBoardingPass() {
    html2canvas(document.getElementById("boardingPassArea")).then(canvas => {
        const link = document.createElement("a");
        link.download = "Boarding_Pass.png";
        link.href = canvas.toDataURL();
        link.click();
    });
}
